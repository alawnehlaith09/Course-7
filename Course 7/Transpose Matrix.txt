#include <iostream>
#include <iomanip>
using namespace std;


void fillArrayWithOrderedNumbers(int arr[3][3], int rows, int cols)
{
	int counter = 0;

	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			arr[i][j] = ++counter;
}

void fillTransposedArray(int arr[3][3], int transposedArr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			transposedArr[i][j] = arr[j][i];
}

void printArray(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
	{
		for (int j = 0; j < cols; j++)
			cout << setw(2) << arr[i][j] << "	";

		cout << endl;
	}
}

void printTransposedArray(int arr[3][3], int rows, int cols)
{
	for (int j = 0; j < cols; j++)
	{
		for (int i = 0; i < rows; i++)
			cout << setw(2) << arr[i][j] << "	";
		
		cout << endl;
	}
}

int main()
{
	srand((unsigned)time(NULL));

	int arr[3][3], transposedArr[3][3];

	fillArrayWithOrderedNumbers(arr, 3, 3);
	cout << "The following is a 3x3 transposed matrix: " << endl;
	printArray(arr, 3, 3);
	cout << endl;

	fillTransposedArray(arr, transposedArr, 3, 3);
	cout << "The following is the transposed matrix: " << endl;
	printArray(transposedArr, 3, 3);
}
