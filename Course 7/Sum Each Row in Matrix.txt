#include <iostream>
#include <iomanip>
using namespace std;

int randomNumber(int from, int to)
{
	int random_number = rand() % (to - from + 1) + from;
	return random_number;
}

void fillArrayWithRandomNumbers(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			arr[i][j] = randomNumber(1, 100);
}

int rowSum(int arr[3][3], int rowNumber, int cols)
{
	int sum = 0;

	for (int j = 0; j < cols; j++)
		sum += arr[rowNumber][j];

	return sum;
}

void printEachSumRow(int arr[3][3], int rows, int cols)
{
	cout << "The following is the sum of each row in the matrix: " << endl;
	for (int i = 0; i < rows; i++)
		cout << "Row " << i + 1 << " Sum = " << rowSum(arr, i, cols) << endl;

}

void printArray(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
	{
		if (!(i == 0))
			cout << endl;

		for (int j = 0; j < cols; j++)
			cout << setw(3) << arr[i][j] << "	";
	}

	cout << endl << endl;
}

int main() 
{
	srand((unsigned)time(NULL));

	int arr[3][3];

	fillArrayWithRandomNumbers(arr, 3, 3);

	cout << "The following is a 3x3 random matrix: " << endl;
	printArray(arr, 3, 3);

	printEachSumRow(arr, 3, 3);
}
