#include <iostream>
#include <string>
using namespace std;

string readString()
{
	cout << "Enter a string: ";
	string str;
	getline(cin, str);
	return str;
}

bool isVowel(char ch)
{
	ch = tolower(ch);

	return ((ch == 'a') || (ch == 'e') || (ch == 'i') || (ch == 'o') || (ch == 'u'));
}

void printAllVowels(string str)
{
	for (int i = 0; i < str.length(); i++)
		if (isVowel(str[i]))
			cout << str[i] << "   ";
}

int main()
{
	string str = readString();
	cout << "\nVowels in String are: ";
	printAllVowels(str);
	cout << endl;
}
