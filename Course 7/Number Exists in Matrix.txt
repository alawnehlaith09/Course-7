#include <iostream>
#include <iomanip>
using namespace std;

int randomNumber(int from, int to)
{
	int random_number = rand() % (to - from + 1) + from;
	return random_number;
}

void fillArrayWithRandomNumbers(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			arr[i][j] = randomNumber(1, 10);
}

bool checkNumberInMatrix(int arr[3][3], int rows, int cols, int number)
{
	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			if (arr[i][j] == number)
				return true;

	return false;
}

void printArray(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
	{
		if (!(i == 0))
			cout << endl;

		for (int j = 0; j < cols; j++)
			cout << arr[i][j] << "	";
	}
}

int main()
{
	srand((unsigned)time(NULL));

	int arr[3][3];

	fillArrayWithRandomNumbers(arr, 3, 3);
	cout << "Matrix: " << endl;
	printArray(arr, 3, 3);
	cout << endl << endl;

	int number;
	cout << "Enter a number to look for in matrix: ";
	cin >> number;
	cout << endl;

	if (checkNumberInMatrix(arr, 3, 3, number))
		cout << "Yes, it is there.\n";
	else
		cout << "No, it is not there.\n";

}
