#include <iostream>
#include <iomanip>
using namespace std;

int randomNumber(int from, int to)
{
	int random_number = rand() % (to - from + 1) + from;
	return random_number;
}

void fillArrayWithRandomNumbers(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			arr[i][j] = randomNumber(1, 10);
}

bool checkNumberInMatrix(int arr[3][3], int rows, int cols, int number)
{
	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			if (arr[i][j] == number)
				return true;

	return false;
}

void intersectedNumbersInMatrices(int arr1[3][3], int arr2[3][3], int rows, int cols)
{
	int number;

	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
		{
			number = arr1[i][j];

			if (checkNumberInMatrix(arr2, 3, 3, number))
				cout << number << "    ";
			else
				continue;
		}

}

void printArray(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
	{
		if (!(i == 0))
			cout << endl;

		for (int j = 0; j < cols; j++)
			cout << arr[i][j] << "	";
	}
}

int main()
{
	srand((unsigned)time(NULL));

	int arr1[3][3], arr2[3][3];

	fillArrayWithRandomNumbers(arr1, 3, 3);
	cout << "Matrix1: " << endl;
	printArray(arr1, 3, 3);
	cout << endl << endl;

	fillArrayWithRandomNumbers(arr2, 3, 3);
	cout << "Matrix2: " << endl;
	printArray(arr2, 3, 3);
	cout << endl << endl;

	cout << "Intersected Numbers are: " << endl;
	intersectedNumbersInMatrices(arr1, arr2, 3, 3);
	cout << endl;
}
