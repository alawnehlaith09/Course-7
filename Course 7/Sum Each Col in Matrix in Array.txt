#include <iostream>
#include <iomanip>
using namespace std;

int randomNumber(int from, int to)
{
	int random_number = rand() % (to - from + 1) + from;
	return random_number;
}

void fillArrayWithRandomNumbers(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			arr[i][j] = randomNumber(1, 100);
}

int colSum(int arr[3][3], int rows, int colNumber)
{
	int sum = 0;

	for (int i = 0; i < rows; i++)
		sum += arr[i][colNumber];

	return sum;
}

void fillEachColSum(int arr[3][3], int sumArr[3], int rows, int cols)
{
	for (int j = 0; j < rows; j++)
		sumArr[j] = colSum(arr, rows, j);
}

void printEachColSum(int sumArr[3], int size)
{
	cout << "The following is the sum of each col in the matrix: " << endl;
	for (int i = 0; i < size; i++)
		cout << "Col " << i + 1 << " Sum = " << sumArr[i] << endl;

}

void printArray(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
	{
		if (!(i == 0))
			cout << endl;

		for (int j = 0; j < cols; j++)
			cout << setw(3) << arr[i][j] << "	";
	}

	cout << endl << endl;
}


int main()
{
	srand((unsigned)time(NULL));

	int arr[3][3];
	int sumArr[3];

	fillArrayWithRandomNumbers(arr, 3, 3);

	cout << "The following is a 3x3 random matrix: " << endl;
	printArray(arr, 3, 3);
	cout << endl;

	fillEachColSum(arr, sumArr, 3, 3);
	printEachColSum(sumArr, 3);
}
