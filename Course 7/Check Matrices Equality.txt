#include <iostream>
#include <iomanip>
using namespace std;

int randomNumber(int from, int to)
{
	int random_number = rand() & (to - from + 1) + from;
	return random_number;
}

void fillArrayWithRandomNumbers(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			arr[i][j] = randomNumber(1, 10);
}

int sumOfArray(int arr[3][3], int rows, int cols)
{
	int sum = 0;

	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			sum += arr[i][j];

	return sum;
}



bool areEqualMatrices(int arr1[3][3], int arr2[3][3], int rows, int cols)
{
	return (sumOfArray(arr1, rows, cols) == sumOfArray(arr2, rows, cols));
}


void printArray(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
	{
		if (!(i == 0))
			cout << endl;

		for (int j = 0; j < cols; j++)
			cout << setw(2) << setfill('0') << arr[i][j] << "	";
	}
}

int main()
{
	srand((unsigned)time(NULL));

	int arr1[3][3], arr2[3][3];

	fillArrayWithRandomNumbers(arr1, 3, 3);
	cout << "Matrix1: " << endl;
	printArray(arr1, 3, 3);
	cout << endl << endl;

	fillArrayWithRandomNumbers(arr2, 3, 3);
	cout << "Matrix2: " << endl;
	printArray(arr2, 3, 3);
	cout << endl << endl;

	if (areEqualMatrices(arr1, arr2, 3, 3))
		cout << "\nYes: both matrices are equal.\n";
	else
		cout << "\nNo: matrices are not equal.\n";
}
