#include <iostream>
#include <iomanip>
using namespace std;

int randomNumber(int from, int to)
{
	int random_number = rand() % (to - from + 1) + from;
	return random_number;
}

void fillArrayWithRandomNumbers(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			arr[i][j] = randomNumber(1, 10);
}

int numberCountInMatrix(int arr[3][3], int rows, int cols, int number)
{
	int count = 0;

	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			if (arr[i][j] == number)
				count++;

	return count;
}


bool isSparseMatrix(int arr[3][3], int rows, int cols)
{
	int size = rows * cols;

	return numberCountInMatrix(arr, 3, 3, 0) >= (size / 2);
}


void printArray(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
	{
		if (!(i == 0))
			cout << endl;

		for (int j = 0; j < cols; j++)
			cout << setw(2) << setfill('0') << arr[i][j] << "	";
	}
}

int main()
{
	srand((unsigned)time(NULL));

	int arr[3][3];

	fillArrayWithRandomNumbers(arr, 3, 3);
	cout << "Matrix: " << endl;
	printArray(arr, 3, 3);
	cout << endl << endl;

	if (isSparseMatrix(arr, 3, 3))
		cout << "\nYes: it is sparse.\n";
	else
		cout << "\nNo: matrix is not sparse.\n";
}
