#include <iostream>
#include <string>
using namespace std;

string readString()
{
	string str;
	cout << "Enter a string: ";
	getline(cin, str);
	return str;
}

void printFirstLetterOfEachWord(string str)
{
	bool is_first_letter = true;

	for (int i = 0; i < str.length(); i++)
	{
		if (str[i] != ' ' && is_first_letter)
			cout << str[i] << endl;
		
		is_first_letter = str[i] == ' ' ? true : false;
	}
}

int main()
{
	srand((unsigned)time(NULL));
	
	string str = readString();
	printFirstLetterOfEachWord(str);
}
