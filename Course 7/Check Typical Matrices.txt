#include <iostream>
#include <iomanip>
using namespace std;

int randomNumber(int from, int to)
{
	int random_number = rand() & (to - from + 1) + from;
	return random_number;
}

void fillArrayWithRandomNumbers(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			arr[i][j] = randomNumber(1, 10);
}


bool areTypicalMatrices(int arr1[3][3], int arr2[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			if (arr1[i][j] != arr2[i][j])
				return false;

	return true;
}


void printArray(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
	{
		if (!(i == 0))
			cout << endl;

		for (int j = 0; j < cols; j++)
			cout << setw(2) << setfill('0') << arr[i][j] << "	";
	}
}

int main()
{
	srand((unsigned)time(NULL));

	int arr1[3][3], arr2[3][3];

	fillArrayWithRandomNumbers(arr1, 3, 3);
	cout << "Matrix1: " << endl;
	printArray(arr1, 3, 3);
	cout << endl << endl;

	fillArrayWithRandomNumbers(arr2, 3, 3);
	cout << "Matrix2: " << endl;
	printArray(arr2, 3, 3);
	cout << endl << endl;

	if (areTypicalMatrices(arr1, arr2, 3, 3))
		cout << "\nYes: both matrices are typical.\n";
	else
		cout << "\nNo: matrices are not typical.\n";
}
