#include <iostream>
using namespace std;

void printFibonacciSeries(int n)
{
	int res = 0;
	int prev2 = 0, prev1 = 1;
	cout << "1	";

	for (int i = 2; i <= n; i++)
	{
		res = prev1 + prev2;
		cout << res << "	";
		prev2 = prev1;
		prev1 = res;
	}
	
	cout << endl;
}

int main()
{
	printFibonacciSeries(10);
}
