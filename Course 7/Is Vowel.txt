#include <iostream>
#include <string>
using namespace std;

char readCharacter()
{
	cout << "Enter a character: ";
	char ch;
	cin >> ch;
	return ch;
}

bool isVowel(char ch)
{
	ch = tolower(ch);

	return ((ch == 'a') || (ch == 'e') || (ch == 'i') || (ch == 'o') || (ch == 'u'));
}

int main()
{
	char ch = readCharacter();
	
	if (isVowel(ch))
		cout << "\nYes: Letter '" << ch << "' is vowel.\n";
	else
		cout << "\nNo: Letter '" << ch << "' is not vowel.\n";
}
