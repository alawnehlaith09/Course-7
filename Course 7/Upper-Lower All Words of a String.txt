#include <iostream>
#include <string>
using namespace std;

string readString()
{
	string str;
	cout << "Enter a string: ";
	getline(cin, str);
	return str;
}

string upperAllWords(string str)
{
	for (int i = 0; i < str.length(); i++)
		str[i] = toupper(str[i]);

	return str;
}

string lowerAllWords(string str)
{
	for (int i = 0; i < str.length(); i++)
		str[i] = tolower(str[i]);

	return str;
}

int main()
{
	srand((unsigned)time(NULL));
	
	string str = readString();
	cout << endl;

	str = upperAllWords(str);
	cout << "String after Upper: " << endl;
	cout << str << endl;

	str = lowerAllWords(str);
	cout << "String after Lower: " << endl;
	cout << str << endl;
}
