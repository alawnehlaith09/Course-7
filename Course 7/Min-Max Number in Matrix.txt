#include <iostream>
#include <iomanip>
using namespace std;

int randomNumber(int from, int to)
{
	int random_number = rand() % (to - from + 1) + from;
	return random_number;
}

void fillArrayWithRandomNumbers(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			arr[i][j] = randomNumber(1, 10);
}

int minNumberInMatrix(int arr[3][3], int rows, int cols)
{
	int min = arr[0][0];

	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			if (arr[i][j] < min)
				min = arr[i][j];

	return min;
}

int maxNumberInMatrix(int arr[3][3], int rows, int cols)
{
	int max = arr[0][0];

	for (int i = 0; i < rows; i++)
		for (int j = 0; j < cols; j++)
			if (arr[i][j] > max)
				max = arr[i][j];

	return max;
}


void printArray(int arr[3][3], int rows, int cols)
{
	for (int i = 0; i < rows; i++)
	{
		if (!(i == 0))
			cout << endl;

		for (int j = 0; j < cols; j++)
			cout << arr[i][j] << "	";
	}
}

int main()
{
	srand((unsigned)time(NULL));

	int arr[3][3];

	fillArrayWithRandomNumbers(arr, 3, 3);
	cout << "Matrix: " << endl;
	printArray(arr, 3, 3);
	cout << endl << endl;

	cout << "Minumum Number is: " << minNumberInMatrix(arr, 3, 3) << endl;
	cout << "Maximum Number is: " << maxNumberInMatrix(arr, 3, 3) << endl;
}
